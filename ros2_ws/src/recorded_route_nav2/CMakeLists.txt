cmake_minimum_required(VERSION 3.8)
project(recorded_route_nav2)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp rclcpp_action rclcpp_lifecycle nav2_core nav2_util nav2_costmap_2d
    nav2_regulated_pure_pursuit_controller nav2_msgs nav_msgs geometry_msgs std_msgs
    tf2_ros tf2_geometry_msgs pluginlib behaviortree_cpp_v3 patrol_interfaces)
  find_package(${dep} REQUIRED)
endforeach()
find_package(nlohmann_json REQUIRED)
find_package(ompl REQUIRED)
set(deps rclcpp rclcpp_action rclcpp_lifecycle nav2_core nav2_util nav2_costmap_2d
    nav2_msgs nav_msgs geometry_msgs std_msgs
    tf2_ros tf2_geometry_msgs pluginlib behaviortree_cpp_v3 patrol_interfaces)
add_library(recorded_route_controller SHARED src/controller.cpp)
target_include_directories(recorded_route_controller PUBLIC include)
ament_target_dependencies(recorded_route_controller ${deps} nav2_regulated_pure_pursuit_controller)
add_library(recorded_route_progress SHARED src/progress.cpp)
target_include_directories(recorded_route_progress PUBLIC include)
ament_target_dependencies(recorded_route_progress nav2_util)
target_link_libraries(recorded_route_controller nlohmann_json::nlohmann_json recorded_route_progress)
# Do not link RPP into a DSO loaded first by the ProgressChecker loader.
# Humble class_loader would associate RPP's factory with the wrong library.
add_library(recorded_route_checkers SHARED src/checkers.cpp)
target_include_directories(recorded_route_checkers PUBLIC include)
ament_target_dependencies(recorded_route_checkers ${deps})
target_link_libraries(recorded_route_checkers nlohmann_json::nlohmann_json recorded_route_progress)
pluginlib_export_plugin_description_file(nav2_core plugins.xml)
add_library(recorded_route_entry SHARED src/entry_planner.cpp)
target_include_directories(recorded_route_entry PUBLIC include ${OMPL_INCLUDE_DIRS})
ament_target_dependencies(recorded_route_entry ${deps})
target_link_libraries(recorded_route_entry nlohmann_json::nlohmann_json ${OMPL_LIBRARIES})
pluginlib_export_plugin_description_file(nav2_core entry_plugins.xml)
add_library(recorded_route_bt SHARED src/follow_recorded_leg.cpp)
target_include_directories(recorded_route_bt PUBLIC include)
target_compile_definitions(recorded_route_bt PRIVATE BT_PLUGIN_EXPORT)
ament_target_dependencies(recorded_route_bt ${deps})
target_link_libraries(recorded_route_bt nlohmann_json::nlohmann_json)
install(TARGETS recorded_route_controller recorded_route_checkers recorded_route_progress
  recorded_route_bt recorded_route_entry LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES plugins.xml entry_plugins.xml README.md DESTINATION share/${PROJECT_NAME})
ament_export_dependencies(${deps})
ament_export_libraries(recorded_route_controller)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(recorded_geometry_test test/test_geometry.cpp)
  target_include_directories(recorded_geometry_test PRIVATE include ${OMPL_INCLUDE_DIRS})
  ament_target_dependencies(recorded_geometry_test nav_msgs geometry_msgs tf2_geometry_msgs)
  target_link_libraries(recorded_geometry_test nlohmann_json::nlohmann_json ${OMPL_LIBRARIES})
  ament_add_gtest(recorded_tracking_test test/test_tracking.cpp)
  target_include_directories(recorded_tracking_test PRIVATE include)
  ament_target_dependencies(recorded_tracking_test nav_msgs geometry_msgs tf2_geometry_msgs)
  target_link_libraries(recorded_tracking_test nlohmann_json::nlohmann_json)
endif()
ament_package()
