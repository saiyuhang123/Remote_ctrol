cmake_minimum_required(VERSION 3.8)
project(msf_localizer)
if(POLICY CMP0074)
  cmake_policy(SET CMP0074 NEW)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
foreach(dep rclcpp sensor_msgs geometry_msgs nav_msgs std_msgs diagnostic_msgs tf2_ros pcl_conversions)
  find_package(${dep} REQUIRED)
endforeach()
find_package(PCL REQUIRED COMPONENTS common io filters registration)
find_package(Eigen3 REQUIRED)
add_executable(multi_sensor_fusion src/multi_sensor_fusion.cpp)
target_include_directories(multi_sensor_fusion PRIVATE include ${PCL_INCLUDE_DIRS})
target_link_libraries(multi_sensor_fusion ${PCL_LIBRARIES} Eigen3::Eigen)
ament_target_dependencies(multi_sensor_fusion rclcpp sensor_msgs geometry_msgs nav_msgs
  std_msgs diagnostic_msgs tf2_ros pcl_conversions)
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(fusion_geometry_test test/test_fusion_geometry.cpp)
  target_include_directories(fusion_geometry_test PRIVATE include)
  target_link_libraries(fusion_geometry_test Eigen3::Eigen)
endif()
install(TARGETS multi_sensor_fusion DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
ament_package()
